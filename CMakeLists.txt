cmake_minimum_required(VERSION 3.20)
project(DragonRider VERSION 0.1.0 LANGUAGES CXX)

add_library(dragonrider_core src/core/binary_reader.cpp)
target_include_directories(dragonrider_core PUBLIC include)
target_compile_features(dragonrider_core PUBLIC cxx_std_20)
set_target_properties(dragonrider_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(dragonrider_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(dragonrider_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

find_package(ZLIB REQUIRED)
add_library(dragonrider_engine src/assets/riot/srsc_header.cpp src/assets/riot/srsc_records.cpp src/assets/riot/srsc_manifest.cpp src/assets/riot/level_directory_0000.cpp src/assets/riot/level_0001.cpp src/assets/riot/level_cell_topology.cpp src/assets/riot/texture_0040.cpp src/assets/riot/texture_0040_rows.cpp src/assets/riot/palette_0030.cpp src/assets/riot/indexed_8.cpp)
target_link_libraries(dragonrider_engine PUBLIC dragonrider_core PRIVATE ZLIB::ZLIB)
target_compile_features(dragonrider_engine PUBLIC cxx_std_20)
set_target_properties(dragonrider_engine PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(dragonrider_engine PRIVATE /W4 /permissive-)
else()
  target_compile_options(dragonrider_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(dragonrider_inspect apps/inspect/main.cpp)
target_link_libraries(dragonrider_inspect PRIVATE dragonrider_engine ZLIB::ZLIB)
set_target_properties(dragonrider_inspect PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(dragonrider_inspect PRIVATE /W4 /permissive-)
else()
  target_compile_options(dragonrider_inspect PRIVATE -Wall -Wextra -Wpedantic)
endif()
